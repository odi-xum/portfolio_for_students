erDiagram
    User ||--o{ Event : создаёт
    Event ||--o{ EventFile : содержит
    User ||--o{ Notification : получает
    User ||--o{ ScholarshipRequest : подаёт
    User ||--o{ CuratorOKOverride : оценивается
    User ||--o{ CuratorOKOverride : оценивает
    User ||--o{ AuditLog : порождает
    User ||--o{ Group : курирует
    Group ||--o{ User : обучает
    Specialty ||--o{ Group : имеет
    Department ||--|{ Specialty : включает

    User {
        int id PK
        string username
        string password_hash
        string role "admin | student | curator | commission"
        string last_name
        string first_name
        string patronymic
        string group_name
        int group_id FK
    }

    Event {
        int id PK
        int student_id FK
        string title
        text description
        string status "pending | approved | rejected | disputed"
        string previous_status
        string category "OK-1 ... OK-9"
        int score "1-5"
        text curator_comment
        text commission_comment
        datetime created_at
        datetime commission_reviewed_at
    }

    EventFile {
        int id PK
        int event_id FK
        string file_path
        string file_type "png | jpg | pdf"
    }

    Notification {
        int id PK
        int user_id FK
        text message
        string link
        bool is_read
        datetime created_at
    }

    ScholarshipRequest {
        int id PK
        int student_id FK
        string status "under_curator_review | under_commission_review | approved | rejected"
        datetime created_at
        datetime commission_reviewed_at
    }

    Department {
        int id PK
        string name
        datetime created_at
    }

    Specialty {
        int id PK
        string name
        string code
        string abbreviation
        int department_id FK
    }

    Group {
        int id PK
        string name
        string group_number
        int specialty_id FK
        string specialty_name
        string specialty_code
        string budget_type "budget | commerce"
        int start_year
        int course
        string department
        string form_of_study
        date start_date
        date end_date
        int curator_id FK
    }

    CuratorOKOverride {
        int id PK
        int student_id FK
        int curator_id FK
        string ok_category "OK-1 ... OK-9"
        datetime created_at
    }

    AuditLog {
        int id PK
        int user_id FK
        string username
        string action
        text details
        string ip_address
        datetime created_at
    }
